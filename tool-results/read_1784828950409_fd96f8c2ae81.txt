    56→// الحصول على الوقت الحالي بصيغة HH:MM
    57→function currentTimeStr(): string {
    58→  const d = new Date()
    59→  return `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`
    60→}
    61→
    62→// استخراج الوقت بصيغة HH:MM من تاريخ ISO
    63→function timeFromISO(iso: string | null): string {
    64→  if (!iso) return ''
    65→  const d = new Date(iso)
    66→  return `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`
    67→}
    68→
    69→// دمج التاريخ والوقت في ISO
    70→function combineDateTime(dateStr: string, timeStr: string): string {
    71→  const d = new Date(dateStr)
    72→  const [h, m] = timeStr.split(':').map(Number)
    73→  d.setHours(h, m, 0, 0)
    74→  return d.toISOString()
    75→}
