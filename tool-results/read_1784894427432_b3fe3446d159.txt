    77→async function fetchOrdersData(statusFilter: string): Promise<OrdersData> {
    78→  const [orders, products, materials] = await Promise.all([
    79→    statusFilter === 'all'
    80→      ? productionOrderRepository.getAll()
    81→      : productionOrderRepository.getByStatus(statusFilter as ProductionOrder['status']),
    82→    productRepository.getAll(),
    83→    materialRepository.getAll(),
    84→  ])
    85→  return {
    86→    orders: orders.sort(
    87→      (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
    88→    ),
    89→    products,
    90→    materials,
    91→  }
    92→}

   290→function ProductionOrderDetail({ order, onBack, onChanged }: DetailProps) {
   291→  const { toast } = useToast()
   292→  const [completeOpen, setCompleteOpen] = useState(false)
   293→
   294→  const handleStartStage = async (stageId: string) => {
   295→    try {
   296→      await productionOrderRepository.startStage(order.id, stageId)
   297→      dataChangeEmitter.notifyUpdate('productionOrders')
   298→      onChanged()
   299→      toast({ title: 'تم بدء المرحلة' })
   300→    } catch (e: any) {
   301→      toast({ title: 'خطأ', description: e.message, variant: 'destructive' })
   302→    }
   303→  }
   304→
   305→  const handleCompleteStage = async (stageId: string) => {
   306→    try {
   307→      await productionOrderRepository.completeStage(order.id, stageId)
   308→      dataChangeEmitter.notifyUpdate('productionOrders')
   309→      onChanged()
   310→      toast({ title: 'تم إكمال المرحلة' })
   311→    } catch (e: any) {
   312→      toast({ title: 'خطأ', description: e.message, variant: 'destructive' })
   313→    }
   314→  }
   315→
   316→  const handleCancel = async () => {
   317→    if (!confirm('إلغاء أمر التشغيل؟')) return
   318→    try {
   319→      await productionOrderRepository.update(order.id, {
   320→        status: 'cancelled',
   321→      })
   322→      dataChangeEmitter.notifyUpdate('productionOrders')
   323→      onChanged()
   324→      toast({ title: 'تم إلغاء الأمر' })
   325→    } catch (e: any) {
   326→      toast({ title: 'خطأ', description: e.message, variant: 'destructive' })
   327→    }
   328→  }

   330→  const handleCompleteOrder = async (completedQuantity: number) => {

   332→      await productionOrderRepository.completeOrder(order.id, completedQuantity)

   339→      toast({ title: 'خطأ', description: e.message, variant: 'destructive' })
   340→    }
   341→  }
   342→
   343→  const progress =
   344→    order.quantity > 0
   345→      ? Math.min(100, (order.completedQuantity / order.quantity) * 100)

   361→          variant="outline"
   362→          className={`text-[10px] ${STATUS_STYLES[order.status]}`}
   363→        >
   364→          {STATUS_LABELS[order.status]}
   365→        </Badge>
   366→      </div>
   367→
   368→      {/* رأس الأمر */}
   369→      <div className="bg-gradient-to-br from-slate-800 to-slate-900 text-white rounded-2xl p-4 shadow-md">
   370→        <div className="flex items-start justify-between mb-3">
   371→          <div>
   372→            <p className="text-[10px] text-slate-300">رقم الأمر</p>
   373→            <p className="text-lg font-bold">{order.orderNumber}</p>
   374→          </div>
   375→          <div className="w-10 h-10 rounded-xl bg-white/10 backdrop-blur flex items-center justify-center">
   376→            <Factory className="w-5 h-5 text-emerald-400" />

   383→          </div>
   384→          <div className="bg-white/10 backdrop-blur rounded-lg p-2">
   385→            <p className="text-slate-300">الكمية المطلوبة</p>

   387→              {order.quantity} {order.unit}

   540→            })}
   541→          </div>
   542→        )}
   543→      </div>
   544→
   545→      {/* ملاحظات */}
   546→      {order.notes && (
   547→        <div className="bg-white rounded-2xl shadow-sm border border-slate-100 p-3">
   548→          <p className="text-xs font-bold text-slate-700 mb-1">ملاحظات</p>
   549→          <p className="text-[11px] text-slate-600 whitespace-pre-wrap">
   550→            {order.notes}
   551→          </p>
   552→        </div>
   553→      )}
   554→
   555→      {/* أزرار الإجراءات */}
   556→      {!isCompleted && !isCancelled && (
   557→        <div className="grid grid-cols-2 gap-2">
   558→          <Button
   559→            variant="outline"
   560→            onClick={handleCancel}
   561→            className="h-9 text-xs font-medium border-rose-200 text-rose-700 hover:bg-rose-50"
   562→          >
   563→            <X className="w-4 h-4 ml-1" />
   564→            إلغاء الأمر
   565→          </Button>
   566→          <Button
   567→            onClick={() => setCompleteOpen(true)}
   568→            className="bg-emerald-600 hover:bg-emerald-700 text-white h-9 text-xs font-medium"
   569→          >
   570→            <CheckCircle2 className="w-4 h-4 ml-1" />
   571→            إكمال الأمر
   572→          </Button>
   573→        </div>
   574→      )}
   575→
   576→      {completeOpen && (
   577→        <CompleteOrderDialog
   578→          open={completeOpen}
   579→          onOpenChange={setCompleteOpen}
   580→          order={order}
   581→          onConfirm={handleCompleteOrder}
   582→        />
   583→      )}
   584→    </div>
   585→  )
   586→}

   674→
   675→function ProductionOrderForm({
   676→  open,
   677→  onOpenChange,
   678→  products,
   679→  materials,
   680→  onSaved,
   681→}: FormProps) {
   682→  const [productId, setProductId] = useState('')
   683→  const [quantity, setQuantity] = useState('1')
   684→  const [selectedMaterials, setSelectedMaterials] = useState<MaterialSelection[]>([])
   685→  const [stages, setStages] = useState<string[]>(DEFAULT_STAGES)
   686→  const [expectedEndDate, setExpectedEndDate] = useState('')
   687→  const [notes, setNotes] = useState('')
   688→  const [saving, setSaving] = useState(false)
   689→  const { toast } = useToast()
   690→
   691→  // reset عند الإغلاق
   692→  useEffect(() => {
   693→    if (!open) {
   694→      // تأخير بسيط لتجنب flicker
   695→      Promise.resolve().then(() => {
   696→        setProductId('')
   697→        setQuantity('1')
   698→        setSelectedMaterials([])
   699→        setStages(DEFAULT_STAGES)
   700→        setExpectedEndDate('')
   701→        setNotes('')
   702→      })
   703→    }
   704→  }, [open])
   705→
   706→  const selectedProduct = products.find((p) => p.id === productId)
   707→
   708→  // المواد الخام المتاحة (في مخازن المواد الخام)
   709→  const availableMaterials = materials
   710→
   711→  const addMaterial = () => {
   712→    setSelectedMaterials([
   713→      ...selectedMaterials,
   714→      { materialId: '', quantity: 1 },
   715→    ])
   716→  }
   717→
   718→  const updateMaterial = (idx: number, field: 'materialId' | 'quantity', value: any) => {
   719→    const updated = [...selectedMaterials]
   720→    updated[idx] = { ...updated[idx], [field]: value }
   721→    setSelectedMaterials(updated)
   722→  }
   723→
   724→  const removeMaterial = (idx: number) => {
   725→    setSelectedMaterials(selectedMaterials.filter((_, i) => i !== idx))
   726→  }
   727→
   728→  const addStage = () => {
   729→    setStages([...stages, ''])
   730→  }
   731→
   732→  const updateStage = (idx: number, value: string) => {
   733→    const updated = [...stages]
   734→    updated[idx] = value
   735→    setStages(updated)
   736→  }
   737→
   738→  const removeStage = (idx: number) => {
   739→    setStages(stages.filter((_, i) => i !== idx))
   740→  }
   741→
   742→  const save = async () => {
   743→    if (!productId) {
   744→      toast({ title: 'اختر المنتج', variant: 'destructive' })
   745→      return
   746→    }
   747→    if (!selectedProduct) {
   748→      toast({ title: 'المنتج غير موجود', variant: 'destructive' })
   749→      return
   750→    }
   751→    const qty = Number(quantity)
   752→    if (!qty || qty <= 0) {
   753→      toast({ title: 'أدخل كمية صحيحة', variant: 'destructive' })
   754→      return
   755→    }
   756→
   757→    // تحقق من المواد
   758→    const validMaterials = selectedMaterials.filter((m) => m.materialId)
   759→    for (const sm of validMaterials) {
   760→      const mat = materials.find((m) => m.id === sm.materialId)
   761→      if (!mat) {
   762→        toast({ title: 'مادة غير موجودة', variant: 'destructive' })
   763→        return
   764→      }
   765→      if (mat.quantity < sm.quantity * qty) {
   766→        toast({
   767→          title: `المادة ${mat.name} غير متاحة`,
   768→          description: `متاح: ${mat.quantity} ${mat.unit} • مطلوب: ${sm.quantity * qty}`,
   769→          variant: 'destructive',
   770→        })
   771→        return
   772→      }
   773→    }
   774→
   775→    const validStages = stages.filter((s) => s.trim())
   776→
   777→    setSaving(true)
   778→    try {
   779→      await productionOrderRepository.createOrder({
   780→        productId: selectedProduct.id,
   781→        productName: selectedProduct.name,
   782→        quantity: qty,
   783→        unit: selectedProduct.unit,
   784→        materials: validMaterials.map((sm) => {
   785→          const mat = materials.find((m) => m.id === sm.materialId)!
   786→          return {
   787→            materialId: sm.materialId,
   788→            materialName: mat.name,
   789→            quantity: sm.quantity * qty,
   790→            unit: mat.unit,
   791→          }
   792→        }),
   793→        stages: validStages.map((name) => ({ name: name.trim() })),
   794→        expectedEndDate: expectedEndDate || undefined,
   795→        notes: notes.trim() || undefined,
   796→      })
   797→      dataChangeEmitter.notifyCreate('productionOrders')
   798→      dataChangeEmitter.notifyUpdate('materials')
   799→      dataChangeEmitter.notifyCreate('materialTransactions')
   800→      toast({
   801→        title: 'تم إنشاء أمر التشغيل',
   802→        description: 'تم سحب المواد الخام من المخزن',
   803→      })
   804→      onSaved()
   805→    } catch (e: any) {
   806→      toast({ title: 'خطأ', description: e.message, variant: 'destructive' })
   807→    } finally {
   808→      setSaving(false)
   809→    }
   810→  }
   811→
   812→  return (
   813→    <Dialog open={open} onOpenChange={onOpenChange}>
   814→      <DialogContent className="max-w-lg max-h-[92vh] overflow-y-auto" dir="rtl">
   815→        <DialogHeader>
   816→          <DialogTitle className="text-right">أمر تشغيل جديد</DialogTitle>
   817→          <DialogDescription className="sr-only">إدارة أوامر التشغيل</DialogDescription>
   818→        </DialogHeader>
   819→        <div className="space-y-3">
   820→          {/* المنتج */}
   821→          <div>
   822→            <Label className="text-xs">المنتج *</Label>
   823→            <Select
   824→              value={productId || '__none__'}
   825→              onValueChange={(v) => setProductId(v === '__none__' ? '' : v)}
   826→            >
   827→              <SelectTrigger className="bg-slate-50">
   828→                <SelectValue placeholder="اختر المنتج" />
   829→              </SelectTrigger>
   830→              <SelectContent>
   831→                <SelectItem value="__none__">— اختر —</SelectItem>
   832→                {products.map((p) => (
   833→                  <SelectItem key={p.id} value={p.id}>
   834→                    {p.name} (متاح: {p.quantity} {p.unit})
   835→                  </SelectItem>
   836→                ))}
   837→              </SelectContent>
   838→            </Select>
   839→          </div>
   840→
   841→          {/* الكمية */}
   842→          <div>
   843→            <Label className="text-xs">الكمية المطلوبة *</Label>
   844→            <Input
   845→              type="number"
   846→              value={quantity}
   847→              onChange={(e) => setQuantity(e.target.value)}
   848→              placeholder="1"
   849→              className="bg-slate-50 font-bold"
   850→            />
   851→            {selectedProduct && (
   852→              <p className="text-[10px] text-slate-500 mt-1">
   853→                التكلفة المتوقعة: {formatCurrency(selectedProduct.cost * (Number(quantity) || 0))}
   854→              </p>
   855→            )}
   856→          </div>
   857→
   858→          {/* المواد الخام */}
   859→          <div className="bg-slate-50 rounded-lg p-2 space-y-2">
   860→            <div className="flex items-center justify-between">
   861→              <Label className="text-xs font-bold">المواد الخام</Label>
   862→              <Button
   863→                type="button"
   864→                variant="outline"
   865→                size="sm"
   866→                onClick={addMaterial}
   867→                className="h-7 text-[11px] border-amber-200 text-amber-700 hover:bg-amber-50"
   868→              >
   869→                <Plus className="w-3 h-3 ml-1" />
   870→                إضافة مادة
   871→              </Button>
   872→            </div>
   873→            {selectedMaterials.length === 0 ? (
   874→              <p className="text-[10px] text-slate-400 text-center py-2">
   875→                لا توجد مواد مضافة (يمكن الإضافة لاحقاً)
   876→              </p>
   877→            ) : (
   878→              <div className="space-y-2">
   879→                {selectedMaterials.map((sm, idx) => {
   880→                  const mat = materials.find((m) => m.id === sm.materialId)
   881→                  return (
   882→                    <div
   883→                      key={idx}
   884→                      className="bg-white rounded-lg p-2 space-y-1.5 border border-slate-200"
   885→                    >
   886→                      <div className="flex items-center gap-2">
   887→                        <Select
   888→                          value={sm.materialId || '__none__'}
   889→                          onValueChange={(v) =>
   890→                            updateMaterial(idx, 'materialId', v === '__none__' ? '' : v)
   891→                          }
   892→                        >
   893→                          <SelectTrigger className="bg-slate-50 text-xs h-8 flex-1">
   894→                            <SelectValue placeholder="اختر المادة" />
   895→                          </SelectTrigger>
   896→                          <SelectContent>
   897→                            <SelectItem value="__none__">— اختر —</SelectItem>
   898→                            {availableMaterials.map((m) => (
   899→                              <SelectItem key={m.id} value={m.id}>
   900→                                {m.name} (متاح: {m.quantity} {m.unit})
   901→                              </SelectItem>
   902→                            ))}
   903→                          </SelectContent>
   904→                        </Select>
   905→                        <Button
   906→                          type="button"
   907→                          variant="ghost"
   908→                          size="icon"
   909→                          onClick={() => removeMaterial(idx)}
   910→                          className="h-7 w-7 text-rose-500 shrink-0"
   911→                        >
   912→                          <Trash2 className="w-3 h-3" />
   913→                        </Button>
   914→                      </div>
   915→                      <div className="grid grid-cols-2 gap-1.5">
   916→                        <div>
   917→                          <Label className="text-[10px]">الكمية لكل وحدة</Label>
   918→                          <Input
   919→                            type="number"
   920→                            value={sm.quantity}
   921→                            onChange={(e) =>
   922→                              updateMaterial(idx, 'quantity', Number(e.target.value))
   923→                            }
   924→                            className="bg-slate-50 text-xs h-8"
   925→                          />
   926→                        </div>
   927→                        <div>
   928→                          <Label className="text-[10px]">الإجمالي</Label>
   929→                          <div className="h-8 px-2 flex items-center bg-amber-50 rounded-md text-[11px] font-bold text-amber-700">
   930→                            {sm.quantity * (Number(quantity) || 0)} {mat?.unit || ''}
   931→                          </div>
   932→                        </div>
   933→                      </div>
   934→                    </div>
   935→                  )
   936→                })}
   937→              </div>
   938→            )}
   939→          </div>
   940→
   941→          {/* مراحل التصنيع */}
   942→          <div className="bg-slate-50 rounded-lg p-2 space-y-2">
   943→            <div className="flex items-center justify-between">
   944→              <Label className="text-xs font-bold flex items-center gap-1">
   945→                <Layers className="w-3 h-3" />
   946→                مراحل التصنيع
   947→              </Label>
   948→              <Button
   949→                type="button"
   950→                variant="outline"
   951→                size="sm"
   952→                onClick={addStage}
   953→                className="h-7 text-[11px] border-blue-200 text-blue-700 hover:bg-blue-50"
   954→              >
   955→                <Plus className="w-3 h-3 ml-1" />
   956→                إضافة مرحلة
   957→              </Button>

   959→            <div className="space-y-1.5">
   960→              {stages.map((s, idx) => (
   961→                <div key={idx} className="flex items-center gap-2">
   962→                  <span className="text-[10px] font-bold text-slate-400 w-4">

   981→                  )}
   982→                </div>
   983→              ))}
   984→            </div>
   985→          </div>
   986→
   987→          {/* تاريخ التسليم */}
   988→          <div>

   990→            <Input
   991→              type="date"

   993→              onChange={(e) => setExpectedEndDate(e.target.value)}
   994→              className="bg-slate-50"
   995→            />
   996→          </div>
   997→
   998→          {/* ملاحظات */}
   999→          <div>
  1000→            <Label className="text-xs">ملاحظات</Label>
  1001→            <Textarea
  1002→              value={notes}
  1003→              onChange={(e) => setNotes(e.target.value)}
  1004→              placeholder="ملاحظات إضافية..."
  1005→              className="bg-slate-50 text-sm"
  1006→              rows={2}
  1007→            />
  1008→          </div>
  1009→        </div>
  1010→
  1011→        <DialogFooter className="gap-2">
  1012→          <Button
  1013→            variant="outline"
  1014→            onClick={() => onOpenChange(false)}
  1015→            disabled={saving}
  1016→            className="h-9 text-xs font-medium"
  1017→          >
  1018→            إلغاء
  1019→          </Button>
  1020→          <Button
  1021→            onClick={save}
  1022→            disabled={saving}
  1023→            className="bg-emerald-600 hover:bg-emerald-700 text-white h-9 text-xs font-medium"
  1024→          >
  1025→            {saving ? 'جارٍ الحفظ...' : 'إنشاء الأمر'}
  1026→          </Button>
  1027→        </DialogFooter>
  1028→      </DialogContent>
  1029→    </Dialog>
  1030→  )
  1031→}
