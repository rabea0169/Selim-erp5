    72→function buildPrintText(data: ReportData, from: string, to: string): string {
    73→  const s = data.summary
    74→  return `تقرير مصنع الملابس
    75→الفترة: ${formatDate(from)} إلى ${formatDate(to)}
    76→----------------------------
    77→المبيعات:     ${formatCurrency(s.salesTotal)}
    78→المشتريات:    ${formatCurrency(s.purchasesTotal)}
    79→المصاريف:     ${formatCurrency(s.expensesTotal)}
    80→سلف العمال:   ${formatCurrency(s.advancesTotal)}
    81→${s.productionTotal > 0 ? `الإنتاج:      ${formatCurrency(s.productionTotal)}\n` : ''}----------------------------
    82→صافي الربح:   ${formatCurrency(s.netProfit)}
    83→----------------------------
    84→${new Date().toLocaleString('ar-EG')}`
    85→}

    87→export function ReportsView() {

    94→
    95→  const load = useCallback(async () => {
    96→    setLoading(true)
    97→    try {
    98→      const [res, workers] = await Promise.all([
    99→        reportRepository.getFullReport(from || undefined, to || undefined),
   100→        workerRepository.getAll(),
   101→      ])
   102→      const workerMap = new Map(workers.map((w) => [w.id, w]))
   103→      // إرفاق اسم العامل بسجلات السلف والقبض والإنتاج والحضور لعرضها في القوائم
   104→      const withWorker = <T extends { workerId: string }>(arr: T[]): (T & { worker?: { id: string; name: string } })[] =>
   105→        arr.map((x) => ({
   106→          ...x,
   107→          worker: workerMap.get(x.workerId)
   108→            ? { id: workerMap.get(x.workerId)!.id, name: workerMap.get(x.workerId)!.name }
   109→            : undefined,
   110→        }))
   111→      const enriched: ReportData = {
   112→        ...res,
   113→        advances: withWorker(res.advances) as any,
   114→        receipts: withWorker(res.receipts) as any,
   115→        productions: withWorker(res.productions) as any,
   116→        attendance: withWorker(res.attendance) as any,
   117→      }
   118→      setData(enriched)

   120→      toast({ title: 'خطأ', description: 'فشل تحميل التقرير', variant: 'destructive' })

   585→function SummaryCard({
   586→  label,
   587→  value,
   588→  icon: Icon,
   589→  color,
   590→  count,
   591→}: {
   592→  label: string
   593→  value: number
   594→  icon: any
   595→  color: 'emerald' | 'rose' | 'amber' | 'purple' | 'blue'
   596→  count: number
   597→}) {
   598→  const colors: Record<string, string> = {
   599→    emerald: 'bg-emerald-50 text-emerald-700 border-emerald-100',

   602→    purple: 'bg-purple-50 text-purple-700 border-purple-100',
   603→    blue: 'bg-blue-50 text-blue-700 border-blue-100',

   606→    <div className={`rounded-xl p-3 border ${colors[color]}`}>
   607→      <div className="flex items-center justify-between mb-1">

   641→function TransactionList({
   642→  items,
   643→  empty,
   644→  titleKey,
   645→  amountKey,
   646→  dateKey,
   647→  color,
   648→  extra,
   649→}: {
   650→  items: any[]
   651→  empty: string
   652→  titleKey: string | ((item: any) => string)
   653→  amountKey: string
   654→  dateKey: string
   655→  color: 'emerald' | 'rose' | 'amber' | 'purple' | 'blue'
   656→  extra?: (item: any) => React.ReactNode
   657→}) {
   658→  const colors: Record<string, string> = {
   659→    emerald: 'text-emerald-700',
   660→    rose: 'text-rose-700',
   661→    amber: 'text-amber-700',
   662→    purple: 'text-purple-700',
   663→    blue: 'text-blue-700',
   664→  }
   665→
   666→  if (items.length === 0) {
   667→    return (
   668→      <div className="bg-white rounded-xl p-6 text-center border border-slate-100">
   669→        <Receipt className="w-8 h-8 text-slate-300 mx-auto mb-2" />
   670→        <p className="text-xs text-slate-500">{empty}</p>
   671→      </div>
   672→    )
   673→  }
   674→
   675→  return (
   676→    <div className="bg-white rounded-xl shadow-sm border border-slate-100 overflow-hidden">
   677→      <div className="max-h-96 overflow-y-auto">
   678→        {items.map((item, i) => {
   679→          const title = typeof titleKey === 'function' ? titleKey(item) : item[titleKey]
   680→          const amount = item[amountKey]
   681→          const date = item[dateKey]
   682→          return (
   683→            <div
   684→              key={item.id || i}
   685→              className="flex items-center justify-between p-3 border-b border-slate-50 last:border-0"
   686→            >
   687→              <div className="flex-1">
   688→                <p className="text-sm font-medium text-slate-800">{title}</p>
   689→                <div className="flex items-center gap-2 mt-0.5">
   690→                  <span className="flex items-center gap-1 text-[10px] text-slate-500">
   691→                    <Calendar className="w-2.5 h-2.5" />
   692→                    {formatDate(date)}
   693→                  </span>
   694→                  {extra && extra(item)}
   695→                </div>
   696→              </div>
   697→              <p className={`text-sm font-bold ${colors[color]}`}>
   698→                {formatCurrency(amount)}
   699→              </p>
   700→            </div>
   701→          )
   702→        })}
   703→      </div>
   704→    </div>
   705→  )
   706→}
