   107→async function fetchWarehousesData(): Promise<WarehousesData> {
   108→  const [warehouses, transactions] = await Promise.all([
   109→    warehouseRepository.getAll(),
   110→    materialTransactionRepository.getAll(),
   111→  ])
   112→  // اجلب كل المواد
   113→  const materials: Material[] = []
   114→  for (const w of warehouses) {
   115→    const ms = await materialRepository.getByWarehouse(w.id)
   116→    materials.push(...ms)
   117→  }
   118→  return {
   119→    warehouses: warehouses.sort((a, b) => a.name.localeCompare(b.name, 'ar')),
   120→    materials,
   121→    transactions,
   122→  }
   123→}
