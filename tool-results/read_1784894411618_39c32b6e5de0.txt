   107→   107→async function fetchWarehousesData(): Promise<WarehousesData> {
   108→   108→  const [warehouses, transactions] = await Promise.all([
   109→   109→    warehouseRepository.getAll(),
   110→   110→    materialTransactionRepository.getAll(),
   111→   111→  ])
   112→   112→  // اجلب كل المواد
   113→   113→  const materials: Material[] = []
   114→   114→  for (const w of warehouses) {
   115→   115→    const ms = await materialRepository.getByWarehouse(w.id)
   116→   116→    materials.push(...ms)
   117→   117→  }
   118→   118→  return {
   119→   119→    warehouses: warehouses.sort((a, b) => a.name.localeCompare(b.name, 'ar')),
   120→   120→    materials,
   121→   121→    transactions,
   122→   122→  }
   123→   123→}

   882→   882→function StockMovementForm({
   883→   883→  open,
   884→   884→  onOpenChange,
   885→   885→  material,
   886→   886→  mode,
   887→   887→  onSaved,
   888→   888→}: StockMovementProps) {
   889→   889→  const isAdd = mode === 'add'
   890→   890→  const [quantity, setQuantity] = useState('')
   891→   891→  const [unitCost, setUnitCost] = useState('')
   892→   892→  const [reason, setReason] = useState(isAdd ? 'شراء' : 'استهلاك')
   893→   893→  const [notes, setNotes] = useState('')
   894→   894→  const [saving, setSaving] = useState(false)
   895→   895→  const { toast } = useToast()
   896→   896→
   897→   897→  // reset عند فتح النموذج
   898→   898→  useEffect(() => {
   899→   899→    if (open) {
   900→   900→      setQuantity('')
   901→   901→      setUnitCost(String(material.unitCost || ''))
   902→   902→      setReason(isAdd ? 'شراء' : 'استهلاك')
   903→   903→      setNotes('')
   904→   904→    }
   905→   905→  }, [open, material.id, isAdd])
   906→   906→
   907→   907→  const save = async () => {
   908→   908→    const num = Number(quantity)
   909→   909→    if (!num || num <= 0) {
   910→   910→      toast({ title: 'أدخل كمية صحيحة', variant: 'destructive' })
   911→   911→      return
   912→   912→    }
   913→   913→    if (isAdd && (!unitCost || Number(unitCost) < 0)) {
   914→   914→      toast({ title: 'أدخل تكلفة الوحدة', variant: 'destructive' })
   915→   915→      return
   916→   916→    }
   917→   917→    if (!isAdd && num > material.quantity) {
   918→   918→      toast({
   919→   919→        title: 'الكمية المطلوبة أكبر من المتاح',
   920→   920→        description: `المتاح: ${material.quantity} ${material.unit}`,
   921→   921→        variant: 'destructive',
   922→   922→      })
   923→   923→      return
   924→   924→    }
   925→   925→    setSaving(true)
   926→   926→    try {
   927→   927→      if (isAdd) {
   928→   928→        await materialRepository.addStock(
   929→   929→          material.id,
   930→   930→          num,
   931→   931→          Number(unitCost) || 0,
   932→   932→          reason,
