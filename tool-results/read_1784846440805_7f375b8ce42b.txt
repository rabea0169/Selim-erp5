    63→    63→// الحصول على الوقت الحالي بصيغة HH:MM
    64→    64→function currentTimeStr(): string {
    65→    65→  const d = new Date()
    66→    66→  return `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`
    67→    67→}
    68→    68→
    69→    69→// استخراج الوقت بصيغة HH:MM من تاريخ ISO
    70→    70→function timeFromISO(iso: string | null): string {
    71→    71→  if (!iso) return ''
    72→    72→  const d = new Date(iso)
    73→    73→  return `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`
    74→    74→}
    75→    75→
    76→    76→// دمج التاريخ والوقت في ISO
    77→    77→function combineDateTime(dateStr: string, timeStr: string): string {
    78→    78→  const d = new Date(dateStr)
    79→    79→  const [h, m] = timeStr.split(':').map(Number)
    80→    80→  d.setHours(h, m, 0, 0)
    81→    81→  return d.toISOString()
    82→    82→}
