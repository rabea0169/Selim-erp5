    57→    57→// الحصول على الوقت الحالي بصيغة HH:MM
    58→    58→function currentTimeStr(): string {
    59→    59→  const d = new Date()
    60→    60→  return `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`
    61→    61→}
    62→    62→
    63→    63→// استخراج الوقت بصيغة HH:MM من تاريخ ISO
    64→    64→function timeFromISO(iso: string | null): string {
    65→    65→  if (!iso) return ''
    66→    66→  const d = new Date(iso)
    67→    67→  return `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`
    68→    68→}
    69→    69→
    70→    70→// دمج التاريخ والوقت في ISO
    71→    71→function combineDateTime(dateStr: string, timeStr: string): string {
    72→    72→  const d = new Date(dateStr)
    73→    73→  const [h, m] = timeStr.split(':').map(Number)
    74→    74→  d.setHours(h, m, 0, 0)
    75→    75→  return d.toISOString()
    76→    76→}
