    99→function buildPrintText(data: ReportData, from: string, to: string): string {
   100→  const s = data.summary
   101→  return `تقرير مصنع الملابس
   102→الفترة: ${formatDate(from)} إلى ${formatDate(to)}
   103→----------------------------
   104→المبيعات:     ${formatCurrency(s.salesTotal)}
   105→المشتريات:    ${formatCurrency(s.purchasesTotal)}
   106→المصاريف:     ${formatCurrency(s.expensesTotal)}
   107→سلف العمال:   ${formatCurrency(s.advancesTotal)}
   108→${s.productionTotal > 0 ? `الإنتاج:      ${formatCurrency(s.productionTotal)}\n` : ''}----------------------------
   109→صافي الربح:   ${formatCurrency(s.netProfit)}
   110→----------------------------
   111→${new Date().toLocaleString('ar-EG')}`
   112→}

   114→export function ReportsView() {
   115→  const [from, setFrom] = useState(startOfMonth())

   122→  const load = async () => {
   123→    setLoading(true)

   125→      const res = await fetch(`/api/reports?from=${from}&to=${to}`).then((r) => r.json())

   133→
   134→  useEffect(() => {

   137→  }, [])

   139→  const setPreset = (preset: 'today' | 'week' | 'month' | 'year') => {
   140→    const now = new Date()
   141→    const today = todayStr()
   142→    let start = ''
   143→    if (preset === 'today') start = today

   178→        )

   180→
   181→      const contentHtml = `

   594→function SummaryCard({
   595→  label,
   596→  value,
   597→  icon: Icon,
   598→  color,
   599→  count,
   600→}: {

   602→  value: number
   603→  icon: any

   606→}) {

   608→    emerald: 'bg-emerald-50 text-emerald-700 border-emerald-100',

   650→function TransactionList({
   651→  items,
   652→  empty,
   653→  titleKey,
   654→  amountKey,
   655→  dateKey,
   656→  color,
   657→  extra,
   658→}: {
   659→  items: any[]
   660→  empty: string
   661→  titleKey: string | ((item: any) => string)
   662→  amountKey: string
   663→  dateKey: string
   664→  color: 'emerald' | 'rose' | 'amber' | 'purple' | 'blue'
   665→  extra?: (item: any) => React.ReactNode
   666→}) {
   667→  const colors: Record<string, string> = {
   668→    emerald: 'text-emerald-700',
   669→    rose: 'text-rose-700',
   670→    amber: 'text-amber-700',
   671→    purple: 'text-purple-700',
   672→    blue: 'text-blue-700',
   673→  }
   674→
   675→  if (items.length === 0) {
   676→    return (
   677→      <div className="bg-white rounded-xl p-6 text-center border border-slate-100">
   678→        <Receipt className="w-8 h-8 text-slate-300 mx-auto mb-2" />
   679→        <p className="text-xs text-slate-500">{empty}</p>
   680→      </div>
   681→    )
   682→  }
   683→
   684→  return (
   685→    <div className="bg-white rounded-xl shadow-sm border border-slate-100 overflow-hidden">
   686→      <div className="max-h-96 overflow-y-auto">
   687→        {items.map((item, i) => {
   688→          const title = typeof titleKey === 'function' ? titleKey(item) : item[titleKey]
   689→          const amount = item[amountKey]
   690→          const date = item[dateKey]
   691→          return (
   692→            <div
   693→              key={item.id || i}
   694→              className="flex items-center justify-between p-3 border-b border-slate-50 last:border-0"
   695→            >
   696→              <div className="flex-1">
   697→                <p className="text-sm font-medium text-slate-800">{title}</p>
   698→                <div className="flex items-center gap-2 mt-0.5">
   699→                  <span className="flex items-center gap-1 text-[10px] text-slate-500">
   700→                    <Calendar className="w-2.5 h-2.5" />
   701→                    {formatDate(date)}
   702→                  </span>
   703→                  {extra && extra(item)}
   704→                </div>
   705→              </div>
   706→              <p className={`text-sm font-bold ${colors[color]}`}>
   707→                {formatCurrency(amount)}
   708→              </p>
   709→            </div>
   710→          )
   711→        })}
   712→      </div>
   713→    </div>
   714→  )
   715→}
